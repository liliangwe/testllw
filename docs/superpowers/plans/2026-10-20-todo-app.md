> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为内部用户提供命令行待办事项记录工具，支持新增事项（名称+描述），数据持久化到 JSON 文件。

**Architecture:** 纯标准库 Python CLI 工具。`TodoItem` dataclass 建模单条事项，`TodoManager` 负责 JSON 文件读写与追加，`argparse` 子命令驱动 CLI。所有逻辑集中在单文件 `todo.py`，测试文件 `test_todo.py` 独立覆盖核心路径。

**Tech Stack:** Python 3.12, stdlib only (argparse, dataclasses, json, pathlib, unittest)

- Test framework: `unittest` (pytest is NOT installed). Run tests with `python3 -m unittest -v`.
- TDD workflow: write the failing test first (RED), then implement to GREEN.

## File Structure


`todo.py` responsibilities:

`test_todo.py` responsibilities:
- `TestTodoItem`: round-trip dict serialization, field defaults.
- `TestTodoManager`: add to empty file, add to existing file, file creation.
- `TestTodoCLI`: run `python3 todo.py add ...` via subprocess, assert stdout/stderr.

```python
from __future__ import annotations

from dataclasses import dataclass, field


@dataclass
class TodoItem:
    name: str
    description: str = ""

    def to_dict(self) -> dict:
        return {"name": self.name, "description": self.description}

    @classmethod
    def from_dict(cls, d: dict) -> TodoItem:
        return cls(name=d["name"], description=d.get("description", ""))
```

```python
import json
from pathlib import Path


class TodoManager:
    def __init__(self, filepath: str) -> None:
        self.filepath = Path(filepath)

    def add(self, item: TodoItem) -> None:
        items: list[dict] = []
        if self.filepath.exists():
            with open(self.filepath, "r", encoding="utf-8") as f:
                items = json.load(f)
        items.append(item.to_dict())
        with open(self.filepath, "w", encoding="utf-8") as f:
            json.dump(items, f, ensure_ascii=False, indent=2)
```

```python
import subprocess


class TestTodoCLI(unittest.TestCase):
    def setUp(self):
        self.tmpdir = tempfile.mkdtemp()
        self.datafile = Path(self.tmpdir) / "todos.json"
        self.env = {**os.environ, "TODO_DATA_FILE": str(self.datafile)}

    def tearDown(self):
        if self.datafile.exists():
            self.datafile.unlink()
        os.rmdir(self.tmpdir)

    def test_add_subcommand_prints_success(self):
        result = subprocess.run(
            ["python3", "todo.py", "add", "--name", "买菜", "--description", "去超市"],
            capture_output=True,
            text=True,
            env=self.env,
        )
        self.assertEqual(result.returncode, 0)
        self.assertIn("已添加待办事项", result.stdout)
        self.assertIn("买菜", result.stdout)

    def test_add_without_description_defaults_empty(self):
        result = subprocess.run(
            ["python3", "todo.py", "add", "--name", "开会"],
            capture_output=True,
            text=True,
            env=self.env,
        )
        self.assertEqual(result.returncode, 0)
        self.assertTrue(self.datafile.exists())
        with open(self.datafile, "r", encoding="utf-8") as f:
            data = json.load(f)
        self.assertEqual(data[0]["description"], "")

    def test_add_missing_name_shows_error(self):
        result = subprocess.run(
            ["python3", "todo.py", "add", "--description", "desc only"],

            text=True,
            env=self.env,
```

```python
import argparse
import os
import sys


def main() -> None:
    parser = argparse.ArgumentParser(description="待办事项记录工具")
    subparsers = parser.add_subparsers(dest="command", required=True)

    add_parser = subparsers.add_parser("add", help="新增待办事项")
    add_parser.add_argument("--name", required=True, help="事项名称")
    add_parser.add_argument("--description", default="", help="事项描述")

    args = parser.parse_args()

    data_file = os.environ.get("TODO_DATA_FILE", str(Path.home() / ".todos.json"))
    mgr = TodoManager(data_file)

    if args.command == "add":
        item = TodoItem(name=args.name, description=args.description)
        mgr.add(item)
        print(f"已添加待办事项: {item.name}")
        if item.description:
            print(f"  描述: {item.description}")


if __name__ == "__main__":
    main()
```
