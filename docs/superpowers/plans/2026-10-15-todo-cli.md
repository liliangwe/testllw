# Todo CLI 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个命令行待办事项工具，支持 `todo add -n <名称> -d <描述>` 将事项追加写入 `~/.todo/tasks.json`。

**Architecture:** 三层结构：`pyproject.toml` 注册入口 → `src/todo_cli/main.py` 负责 argparse 解析与数据层调用 → 数据层使用 `json` + `pathlib` 读写 JSON 文件，原子写入（临时文件 + rename）。测试使用标准库 `unittest`。

**Tech Stack:** Python 3.9+，标准库（argparse / json / pathlib / unittest / tempfile），零第三方依赖。

## Global Constraints

- 语言：Python 3.9+
- 依赖：仅标准库（运行时和测试均零第三方依赖）
- 数据路径：`~/.todo/tasks.json`
- 命令入口：`todo`，通过 `pyproject.toml` 的 `[project.scripts]` 注册
- 最小闭环：仅 `add` 子命令，不包含查看/编辑/删除
- 存储行为：读→追加→写回，原子写入（临时文件+rename）
- 事项名称不能为空、不能包含换行等控制字符
- 退出码：成功 0，参数缺失/非法 1，文件系统错误 1

```python
import json
import os
import string
from datetime import datetime, timezone
from pathlib import Path
from tempfile import NamedTemporaryFile

DEFAULT_DATA_DIR = Path.home() / ".todo"
DEFAULT_DATA_FILE = DEFAULT_DATA_DIR / "tasks.json"


def validate_name(name: str) -> None:
    """校验事项名称：非空且不含控制字符。"""
    if not name:
        raise ValueError("事项名称不能为空")
    if any(c in name for c in string.whitespace):
        raise ValueError("事项名称不能包含空白字符")


def load_tasks(filepath: Path) -> list[dict]:
    """从 JSON 文件加载任务列表，文件不存在时返回空列表。"""
    if not filepath.exists():
        return []
    return json.loads(filepath.read_text(encoding="utf-8"))


def save_tasks(filepath: Path, tasks: list[dict]) -> None:
    """原子写入：先写到同目录临时文件，再 os.replace。"""
    filepath.parent.mkdir(parents=True, exist_ok=True)
    tmp = NamedTemporaryFile(
        mode="w",
        encoding="utf-8",
        dir=filepath.parent,
        delete=False,
        suffix=".tmp",
    )
    try:
        json.dump(tasks, tmp, ensure_ascii=False, indent=2)
        tmp.flush()
        os.fsync(tmp.fileno())
    finally:
        tmp.close()
    os.replace(tmp.name, filepath)


def add_task(
    name: str,
    description: str = "",
    data_file: Path | None = None,
) -> dict:
    """新增一条待办事项。加载→验证→追加→原子保存，返回新任务 dict。"""
    validate_name(name)

    if data_file is None:
        data_file = DEFAULT_DATA_FILE

    task: dict = {
        "name": name,
        "description": description,
        "created_at": datetime.now(timezone.utc).isoformat(),
    }

    tasks = load_tasks(data_file)
    tasks.append(task)
    save_tasks(data_file, tasks)
    return task
```

```python
import argparse
import sys

# ... 保留上方所有数据层代码（imports / constants / validate_name / load_tasks / save_tasks / add_task）...


def build_parser() -> argparse.ArgumentParser:
    """构建命令行参数解析器。"""
    parser = argparse.ArgumentParser(
        prog="todo",
        description="日常待办事项管理工具",
    )
    subparsers = parser.add_subparsers(dest="command")
    add_parser = subparsers.add_parser("add", help="新增待办事项")
    add_parser.add_argument(
        "-n", "--name",
        required=True,
        help="事项名称",
    )
    add_parser.add_argument(
        "-d", "--description",
        default="",
        help="事项描述",
    )
    return parser


def main() -> None:
    """CLI 入口。"""
    parser = build_parser()
    args = parser.parse_args()

    if args.command != "add":
        parser.print_help(sys.stderr)
        sys.exit(1)

    try:
        task = add_task(name=args.name, description=args.description)
    except ValueError as exc:
        print(f"❌ {exc}", file=sys.stderr)
        sys.exit(1)

    print(f"✅ 已添加待办: {task['name']}")
```
